"""Gate V4 H2/H3 under PROTOCOL-v4 (NV1, NV2) with Vera's resumption conditions R1-R3.

Usage:
  python -I fd_checks_v4.py STUDY4_DIR        # writes logs/gate_H2H3_v4.txt and hessians/fd/fd_checks_v4.csv

Reads only archived files: hessians/fd/<d>.out, <d>.dirs, hessians/<d>/{H_R_fwd.bin,y.bin,meta.txt}.
No FD or OTI evaluation is made. driver/fd_checks.py (v3 analysis) is not changed.

R1: input SHA-256 values from REVIEW-vera-v4.md section 4 are verified first; any mismatch stops (exit 2).
R2: a row whose minimum error is at the largest step h_1 has no truncation-side pair and FAILS its slope gate
    (H2, H3 a/b/c); "not resolvable" in H3(c) needs >= 1 truncation-side pair, all failing resolvability.
R3: Vera's section-1 table is reproduced to the printed digits; any mismatch is flagged and the gate is
    reported "STOP (R3 disagreement)" (exit 3).
Readings (PROTOCOL-v4 2.2): (i) truncation side = steps up to and including the minimum; (ii) rho over k in {3,4}
(lenient, Vera F4); (iii) (b) relative to |0.5 v.(g+ + g-)| at the minimising step; (iv) signed descriptive errors.
"""

def sha(p: Path) -> str:
    """Return the SHA-256 hex digest of a file."""
    return hashlib.sha256(p.read_bytes()).hexdigest()


def same_printed(mine: float, printed: str) -> bool:
    """True if `mine` rounds to the value printed by Vera at her printed precision."""
    if "e" in printed:
        sig = len(printed.split("e")[0].lstrip("-").replace(".", ""))
        return float(f"{mine:.{sig - 1}e}") == float(printed)
    dec = len(printed.split(".")[1]) if "." in printed else 0
    return round(mine, dec) == float(printed)


def slope(e: list[float]) -> tuple[bool, bool, int, list[float]]:
    """Return (primary, strict, argmin, ratios). R2: argmin at h_1 -> empty pair set -> primary False."""
    k = int(np.argmin(e))
    rat = [e[i] / e[i + 1] for i in range(k)]
    ok = [r >= 30 for r in rat]
    strict = any(ok[i] and ok[i + 1] for i in range(len(ok) - 1))
    return any(ok), strict, k, rat


def read_out(path: Path) -> tuple[float, dict]:
    """Parse an archived FD output file into R0 and per-label step records."""
    lines = path.read_text().splitlines()
    R0 = float(lines[0].split()[1])
    data: dict = {}
    for ln in lines[1:]:
        t = ln.split()
        data.setdefault(t[0], []).append(
            (float(t[1]), float(t[2]), float(t[3]), np.array(t[4:4 + N], float), np.array(t[4 + N:4 + 2 * N], float)))
    return R0, data


def main(root: Path) -> int:
    """Run the v4 H2/H3 analysis; return the process exit status."""
    out: list[str] = []
    P = out.append
    P("# gate_H2H3_v4 (PROTOCOL-v4 NV1/NV2 + Vera R1-R3); driver/fd_checks_v4.py sha256 "
      + sha(Path(__file__)))
    # ---- R1 and FD output hashes
    bad = []
    for rel, h in R1.items():
        got = sha(root / "hessians" / rel)
        P(f"R1 {rel} {got} {'OK' if got == h else 'MISMATCH'}")
        if got != h:
            bad.append(rel)
    for d, h in FD_OUT.items():
        got = sha(root / "hessians" / "fd" / f"{d}.out")
        P(f"v4s9 fd/{d}.out {got} {'OK' if got == h else 'MISMATCH'}")
        if got != h:
            bad.append(d + ".out")
    if bad:
        P("R1/v4s9: HASH MISMATCH -> STUDY STOPS: " + ", ".join(bad))
        (root / "logs" / "gate_H2H3_v4.txt").write_text("\n".join(out) + "\n")
        print("\n".join(out))
        return 2
    P("R1 PASS: all 20 pinned inputs match; v4 s9 FD output hashes match.")

    v = np.random.default_rng(4).standard_normal(N)
    v /= np.linalg.norm(v)
    rows = []
    r3 = []
    gate_ok = True

    # ---- H2 (NV1)
    for d in H2D:
        H = np.fromfile(root / "hessians" / d / "H_R_fwd.bin", dtype="<f8").reshape(N, N)
        y = np.fromfile(root / "hessians" / d / "y.bin", dtype="<f8")
        _, data = read_out(root / "hessians" / "fd" / f"{d}.out")
        fdc = {}
        for lab, L in data.items():
            if not lab.startswith("col"):
                continue
            i = int(lab[3:])
            ref = H[:, i]
            es = []
            for k, (s, _, _, gp, gm) in enumerate(L):
                assert s == float(f"{HS[k] * max(1.0, abs(y[i])):.17g}"), (d, lab, s)
                c = (gp - gm) / (2 * s)
                es.append(float(np.linalg.norm(c - ref) / np.linalg.norm(ref)))
                fdc.setdefault(i, []).append(c)
            p, st, k, rat = slope(es)
            ok = p and es[k] <= TOL
            gate_ok &= ok
            rows.append(dict(gate="H2", design=d, dir=lab, e=";".join(f"{x:.3e}" for x in es),
                             ratios=";".join(f"{x:.1f}" for x in rat), argmin=k + 1, slope=p, strict_desc=st,
                             best=f"{es[k]:.3e}", status="PASS" if ok else "FAIL"))
        ks = sorted(fdc)
        for a in ks:
            for b in ks:
                if a < b:
                    rs = [abs(fdc[a][j][b] - fdc[b][j][a]) / abs(H[a, b]) for j in range(4)]
                    rows.append(dict(gate="H2-sym(desc)", design=d, dir=f"{a},{b}",
                                     e=";".join(f"{x:.3e}" for x in rs), ratios="", argmin="", slope="",
                                     strict_desc="", best=f"{min(rs):.3e}", status="desc"))

    # ---- H3 (NV2)
    for d in H3D:
        H = np.fromfile(root / "hessians" / d / "H_R_fwd.bin", dtype="<f8").reshape(N, N)
        y = np.fromfile(root / "hessians" / d / "y.bin", dtype="<f8")
        R0, data = read_out(root / "hessians" / "fd" / f"{d}.out")
        dirs = (root / "hessians" / "fd" / f"{d}.dirs").read_text().splitlines()
        vd = np.array([ln.split()[2:] for ln in dirs if ln.startswith("v4")][0], float)
        assert np.array_equal(vd, v), "dirs v != rng(4) v"
        L = data["v4"]
        sc = max(1.0, abs(float(v @ y)))
        Hv = H @ v
        vHv = float(v @ Hv)
        ea, cabs, crel, ec, pg, sds = [], [], [], [], [], []
        ss = []
        for k, (s, Rp, Rm, gp, gm) in enumerate(L):
            assert s == float(f"{HS[k] * sc:.17g}"), (d, s)
            ss.append(s)
            ea.append(float(np.linalg.norm((gp - gm) / (2 * s) - Hv) / np.linalg.norm(Hv)))
            phi = 0.5 * float(v @ (gp + gm))
            cc = abs((Rp - Rm) / (2 * s) - phi)
            cabs.append(cc)
            crel.append(cc / abs(phi))
            sd = (Rp - 2 * R0 + Rm) / s ** 2
            sds.append(sd)
            ec.append(abs(sd - vHv) / abs(vHv))
            pg.append((float(v @ (gp - gm)) / (2 * s) - vHv) / vHv)
        # (a)
        pa, sa, ka, ra = slope(ea)
        oka = pa and ea[ka] <= TOL
        # (b): slope on absolute c; tolerance relative to |phi'| at the minimising step (reading iii)
        pb, sb, kb, rb = slope(cabs)
        okb = pb and crel[kb] <= TOL
        # (c)
        kc = int(np.argmin(ec))
        okc_tol = ec[kc] <= TOL

        def rho(s: float) -> float:
            return max(ec[2] * (ss[2] / s) ** 2, ec[3] * (ss[3] / s) ** 2)

        pairs = list(range(kc))  # truncation-side pairs (k, k+1), 0-based
        resolv = [rho(ss[k + 1]) <= ec[k] / 30 for k in pairs]
        rc = [ec[k] / ec[k + 1] for k in pairs]
        if not pairs:
            cstat, okc_slope = "FAIL (min at h_1; R2)", False
        elif any(resolv):
            okc_slope = any(rc[k] >= 30 for k in pairs if resolv[k])
            cstat = "resolvable; slope " + ("PASS" if okc_slope else "FAIL")
        else:
            cstat, okc_slope = "slope not resolvable by the R reference", True
        okc = okc_tol and okc_slope
        rich = (sds[0] - vHv) / (pg[0] * vHv)
        for nm, E, p, st, k, rat, ok in (("H3a-Hv", ea, pa, sa, ka, ra, oka), ("H3b-c(abs)", cabs, pb, sb, kb, rb, okb)):
            rows.append(dict(gate=nm, design=d, dir="v4", e=";".join(f"{x:.3e}" for x in E),
                             ratios=";".join(f"{x:.1f}" for x in rat), argmin=k + 1, slope=p, strict_desc=st,
                             best=f"{E[k]:.3e}" + (f" rel {crel[k]:.3e}" if nm.startswith("H3b") else ""),
                             status="PASS" if ok else "FAIL"))
        rows.append(dict(gate="H3c-vHv", design=d, dir="v4", e=";".join(f"{x:.3e}" for x in ec),
                         ratios=";".join(f"{x:.1f}" for x in rc), argmin=kc + 1, slope=cstat,
                         strict_desc=slope(ec)[1], best=f"{ec[kc]:.3e}",
                         status="PASS" if okc else "FAIL"))
        rows.append(dict(gate="H3-desc", design=d, dir="v4",
                         e="c_rel " + ";".join(f"{x:.2e}" for x in crel) + " | pg_err " + ";".join(f"{x:.3e}" for x in pg),
                         ratios="rho(s2..s4) " + ";".join(f"{rho(s):.2e}" for s in ss[1:]),
                         argmin="", slope="", strict_desc="", best=f"Richardson r(s1)={rich:.5f}; |v.y|={abs(float(v @ y)):.3f}",
                         status="desc"))
        gate_ok &= oka and okb and okc
        # ---- R3 comparison against Vera's printed digits
        V = VERA[d]
        for nm, mine, theirs in (("e_a", ea, V["ea"]), ("c", crel, V["c"]), ("e_c", ec, V["ec"]), ("pg", pg, V["pg"])):
            for k, t in enumerate(theirs):
                if t is not None:
                    r3.append((d, f"{nm}(h{k + 1})", mine[k], t, same_printed(mine[k], t)))
        r3.append((d, "rho(s2)", rho(ss[1]), V["rho"], same_printed(rho(ss[1]), V["rho"])))
        r3.append((d, "e_c(h1)/30", ec[0] / 30, V["thr"], same_printed(ec[0] / 30, V["thr"])))
        res_mine = bool(resolv[0]) if pairs else False
        r3.append((d, "pair(h1,h2) resolvable", res_mine, str(V["res"]), res_mine == V["res"]))
        if V["fac"] is not None:
            r3.append((d, "e_c(h1)/e_c(h2)", rc[0], V["fac"], same_printed(rc[0], V["fac"])))
        r3.append((d, "Richardson r(s1)", rich, V["rich"], same_printed(rich, V["rich"])))
        r3.append((d, "row pass (a,b,c) [Vera: pass]", oka and okb and okc, "True", oka and okb and okc))

    P("")
    P("## H2 (NV1) and H3 (NV2 with R2); h_1..h_4 = 1e-2..1e-5")
    keys = list(rows[0].keys())
    P(" | ".join(keys))
    for r in rows:
        P(" | ".join(str(r[k_]) for k_ in keys))
    P("")
    P("## R3: reproduction of REVIEW-vera-v4.md section 1 (mine at full precision | Vera printed | match)")
    for d, item, mine, t, m in r3:
        ms = f"{mine:.6e}" if isinstance(mine, float) else str(mine)
        P(f"R3 {d} {item}: {ms} | {t} | {'MATCH' if m else 'MISMATCH'}")
    n_mis = sum(1 for x in r3 if not x[4])
    P(f"R3: {len(r3) - n_mis}/{len(r3)} items match" + ("" if n_mis == 0 else f"; {n_mis} MISMATCH -> STOP for review"))
    P("")
    P("Carried over from the v3 execution (v4 s9 step 5; logs not regenerated): H1a PASS, H1b PASS (logs/step0_analysis.txt).")
    verdict = "PASS" if gate_ok and n_mis == 0 else ("STOP (R3 disagreement)" if gate_ok else "FAIL")
    P(f"GATE V4 (H1a, H1b carried; H2 NV1; H3 a, b, c NV2 + R2; R1; R3) under PROTOCOL-v4: {verdict}")
    (root / "logs" / "gate_H2H3_v4.txt").write_text("\n".join(out) + "\n", encoding="utf-8")
    with open(root / "hessians" / "fd" / "fd_checks_v4.csv", "w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=keys)
        w.writeheader()
        for r in rows:
            w.writerow(r)
    print("\n".join(out))
    return 0 if verdict == "PASS" else 3


if __name__ == "__main__":
    sys.exit(main(Path(sys.argv[1])))
